// IMPORTANTE: en un proyecto de Firebase las reglas viven TODAS en este
// único cuadro (no se pueden separar en dos paneles). Lo que SÍ queda

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ==================== RADIO SIKODARK ====================

    // === USUARIOS ===
    match /usuarios/{docId} {
      allow read, write: if request.auth != null;
    }

    // === VISITAS, ESCUCHAS, MEGUSTAS, TOKENS ===
    match /visitas/{docId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    match /escuchas/{docId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    match /megustas/{docId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    match /tokens/{docId} {
      allow read, write: if request.auth != null;
    }

    // === PRESENCIA ===
    match /presencia/{docId} {
      allow read, write: if request.auth != null;
    }

    // === IMAGENES (la habías dejado afuera — sin este bloque quedaba bloqueada) ===
    match /imagenes/{docId} {
      allow read, write: if request.auth != null;
    }

    // === CONFIG (avisos, publicidad, encuesta) ===
    // Cualquier usuario autenticado (incluido anónimo) puede LEER.
    // Solo el admin puede ESCRIBIR.
    match /config/{docId} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.email == "sikoradio@hotmail.com";
      // Excepción: los votos de la encuesta los puede sumar cualquier oyente
      // (lo hacemos via FieldValue.arrayUnion desde el cliente)
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data)
            .affectedKeys().hasOnly(['encuesta']);
    }

    // === PEDIDOS DE TEMAS ===
    // Cualquier usuario autenticado (incluido anónimo) puede crear pedidos.
    // Solo el admin puede leerlos y borrarlos.
    match /temas_pedidos/{docId} {
      allow create: if request.auth != null
        && request.resource.data.banda is string
        && request.resource.data.banda.size() <= 60
        && request.resource.data.fecha is int;
      allow read, delete, update: if request.auth.token.email == "sikoradio@hotmail.com";
    }

    // ==================== SIKOCHAT (el chat) ====================
    // ⬇️ ADENTRO de match /databases/... — cada bloque independiente de la radio
    match /chat_usuarios/{doc} {
      allow read, write: if true;
    }

    match /chat_mensajes/{doc} {
      allow read, write: if true;
    }

    match /chat_juegos/{doc} {
      allow read, write: if true;
    }

    // === Bloquear todo lo demás por defecto ===
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
